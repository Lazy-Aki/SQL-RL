    // Prepared Statement prevents AST injection while executing ACID transaction
async function fetchDepartmentRollup(db, minBudgetUsd) {
  const sql = `
    SELECT d.dept_name, COUNT(e.id) AS staff_count, AVG(e.salary) AS avg_comp
    FROM departments AS d
    LEFT JOIN employees AS e ON d.dept_name = e.department
    WHERE d.budget_usd >= $1
    GROUP BY d.dept_name
    ORDER BY avg_comp DESC;
  `;
  return await db.executePrepared(sql, [minBudgetUsd]);
}
